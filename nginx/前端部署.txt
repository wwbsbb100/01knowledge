步骤 1：创建前端文件目录
# 创建自定义目录（避免覆盖默认页面）
sudo mkdir -p /var/www/ip-project

# 赋予普通用户写入权限（方便上传文件）
sudo chown -R $USER:$USER /var/www/ip-project

步骤 2：上传前端文件

步骤 3：配置 Nginx 站点
sudo vim /etc/nginx/sites-available/ip-project.conf

server {
    listen 80;
    # 填你的服务器公网IP，有域名就填域名
    server_name 你的服务器IP;

    # 前端文件根目录
    root /var/www/ip-project;
    # 默认首页
    index index.html;

    # 解决前端页面刷新404（可选）
    location / {
        try_files $uri $uri/ /index.html;
    }

    # 反向代理到 Node.js 后端（如果你搭了获取真实IP的Node服务）
    # 假设 Node 服务运行在 3000 端口
    location /api/ip {
        proxy_pass http://127.0.0.1:3000;
        # 传递用户真实IP给后端（核心）
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header Host $host;
    }

    # 日志配置（方便排查问题）
    access_log /var/log/nginx/ip-project_access.log;
    error_log /var/log/nginx/ip-project_error.log;
}

步骤 4：启用站点 + 验证配置
